3. Verification API (Backend)
Create an endpoint to process the token when the user clicks the link. 

javascript
async function verifyEmail(req, res) {
    const { token } = req.query;

    try {
        // 1. Find user with matching token
        const [user] = await db.query('SELECT * FROM users WHERE verification_token = ?', [token]);
        
        if (!user) {
            return res.status(400).json({ error: "Invalid or expired verification token." });
        }

        // 2. Activate user and clear token
        await db.query(
            'UPDATE users SET is_verified = true, verification_token = NULL WHERE id = ?',
            [user.id]
        );

        return res.status(200).json({ message: "Email verified successfully! You can now log in." });
    } catch (error) {
        return res.status(500).json({ error: "Verification failed." });
    }
}
